{{- /* Renders a Compromise Canvas export as an interactive SVG diagram, using the canvas's own node coordinates and edges — so the picture in the post is the picture in the tool. {{< canvas key="valleyrat_shellcode_loader_overview" height="470" >}} Reads site.Data.attackpaths.. Nodes carry position{x,y} but no dimensions, because CompromiseCanvas draws fixed-size cards; the export's 330×200 grid spacing implies a 280×150 card, which is what is drawn here. Edges are coloured by data.actionType (the tactic), and a legend is emitted for the tactics actually present rather than all of them. These graphs can be very wide — the ValleyRAT execution chain is over 7000px — so the SVG opens showing full height and roughly a screen's worth of width, and can be dragged and zoomed. With JavaScript off it degrades to a static, scaled diagram. Params: key (required) data file under data/attackpaths/ height viewport height in px (default 470) download optional URL to the canvas JSON */ -}} {{- $key := .Get "key" -}} {{- $c := index site.Data.attackpaths $key -}} {{- if not $c -}}{{- errorf "canvas: no canvas data for key %q" $key -}}{{- end -}} {{- $h := int (.Get "height" | default 470) -}} {{- $CW := 280 -}}{{- $CH := 150 -}}{{- $PAD := 46 -}} {{- /* index positions and data by node id so edges can be resolved */ -}} {{- $pos := dict -}}{{- $nd := dict -}}{{- $groups := slice -}}{{- $cards := slice -}} {{- range $c.diagram.nodes -}} {{- $pos = merge $pos (dict .id .position) -}} {{- $nd = merge $nd (dict .id .data) -}} {{- if eq .type "labeledGroupNode" -}}{{- $groups = $groups | append . -}} {{- else -}}{{- $cards = $cards | append . -}}{{- end -}} {{- end -}} {{- /* bounds over cards and group frames */ -}} {{- $minX := 1e9 -}}{{- $minY := 1e9 -}}{{- $maxX := -1e9 -}}{{- $maxY := -1e9 -}} {{- range $cards -}} {{- $minX = math.Min $minX .position.x -}}{{- $minY = math.Min $minY .position.y -}} {{- $maxX = math.Max $maxX (add .position.x $CW) -}}{{- $maxY = math.Max $maxY (add .position.y $CH) -}} {{- end -}} {{- range $groups -}} {{- $minX = math.Min $minX .position.x -}}{{- $minY = math.Min $minY .position.y -}} {{- $maxX = math.Max $maxX (add .position.x (.data.width | default 400)) -}} {{- $maxY = math.Max $maxY (add .position.y (.data.height | default 300)) -}} {{- end -}} {{- $vx := sub $minX $PAD -}}{{- $vy := sub $minY $PAD -}} {{- $vw := add (sub $maxX $minX) (mul $PAD 2) -}}{{- $vh := add (sub $maxY $minY) (mul $PAD 2) -}} {{- /* Open on a window rather than the whole graph. Fitting a 7000px-wide diagram into an 780px column scales node text down to ~7px, which is unreadable; capping the initial view keeps labels legible and leaves the sprawl to the "fit" button. */ -}} {{- $ih := int (math.Min $vh 720) -}} {{- $iw := int (math.Min $vw (mul $ih 1.55)) -}} {{- $tactics := slice -}} {{- range $c.diagram.edges -}} {{- with .data -}}{{- with .actionType -}} {{- if not (in $tactics .) -}}{{- $tactics = $tactics | append . -}}{{- end -}} {{- end -}}{{- end -}} {{- end -}} {{- /* only label edges when the graph is sparse; on a 46-edge chain labels collide */ -}} {{- $showEdgeLabels := le (len $c.diagram.edges) 8 -}} {{- $uid := printf "cv-%s" (substr (md5 $key) 0 8) -}} {{- if not (.Page.Scratch.Get "canvasJS") }}{{ .Page.Scratch.Set "canvasJS" true }} {{- end }}
